cmake_minimum_required(VERSION 3.23)
project(Chienpo VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(CHIENPO_BUILD_NATIVE "Build the GLFW/OpenGL/ImGui/OpenAL application" ON)
option(CHIENPO_BUILD_HELIOS "Build the vendored USB/network Helios SDK" ON)
option(CHIENPO_BUILD_TESTS "Build headless game and persistence tests" ON)
include(CTest)

add_library(chienpo_core
    src/types.cpp src/game.cpp src/gameplay.cpp src/render.cpp src/vector_math.cpp
    src/persistence.cpp
    third_party/cjson/cJSON.c)
target_include_directories(chienpo_core PUBLIC include PRIVATE third_party/cjson)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(chienpo_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(CHIENPO_BUILD_NATIVE)
    include(cmake/NativeDependencies.cmake)
    add_executable(chienpo src/main.cpp src/ui.cpp src/audio.cpp src/helios.cpp)
    target_link_libraries(chienpo PRIVATE chienpo_core ad::graphics ad::imguiui ad::sounds ad::soundui ${CMAKE_DL_LIBS})
    target_compile_definitions(chienpo PRIVATE CHIENPO_RESOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/resources")
    if(CHIENPO_BUILD_HELIOS)
        include(cmake/Helios.cmake)
        add_dependencies(chienpo HeliosDACAPI)
        target_compile_definitions(chienpo PRIVATE CHIENPO_HELIOS_LIBRARY_PATH="$<TARGET_FILE:HeliosDACAPI>")
        set_target_properties(chienpo PROPERTIES BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN/../lib")
    endif()
    install(TARGETS chienpo RUNTIME DESTINATION bin)
    install(DIRECTORY resources/ DESTINATION share/chienpo)
endif()

if(CHIENPO_BUILD_TESTS AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/core_tests.cpp")
    add_executable(chienpo_core_tests tests/core_tests.cpp)
    target_link_libraries(chienpo_core_tests PRIVATE chienpo_core)
    target_compile_definitions(chienpo_core_tests PRIVATE CHIENPO_RESOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/resources")
    add_test(NAME chienpo_core COMMAND chienpo_core_tests)
    set_tests_properties(chienpo_core PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")

    add_executable(chienpo_vector_math_tests tests/vector_math_tests.cpp)
    target_link_libraries(chienpo_vector_math_tests PRIVATE chienpo_core)
    add_test(NAME chienpo_vector_math COMMAND chienpo_vector_math_tests)
endif()
